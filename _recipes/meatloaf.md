prep_time: "15 min"

total_time: "1 hr 15 min"
servings: "1 loaf, 6 slices"
image: "https://images.unsplash.com/photo-1705945021699-7374c3fd34cb?auto=format&fit=crop&w=1200&q=80"

ingredient_sections:
  - section_title: "Meatloaf"
    items:
      - "1 1/4 lb ground beef"
      - "1 egg"
      - "3/4 onion, chopped"
      - "3/4 cup milk"
      - "3/4 cup dried bread crumbs"
      - "Salt and pepper to taste"
  - section_title: "Topping"
    items:
      - "1 tbsp + 1 1/2 tsp brown sugar"
      - "1 tbsp + 1 1/2 tsp mustard"
      - "1/4 cup ketchup"

instructions:
  - title: "Mix the Meatloaf"
    text: "Preheat the oven to 350°F. Combine the beef, egg, onion, milk, and bread crumbs, and season with salt and pepper. Place in a 5x9 pan."
  - title: "Add the Topping"
    text: "Combine the brown sugar, mustard, and ketchup. Pour over the meatloaf when 15 min of cooking time is left."
  - title: "Bake"
    text: "Bake at 350°F for 1 hour. Add the topping when 15 min of cook time is left"

notes: "I make this with 1 lb of ground beef, keeping all other measurements the same. I only add ketchup as the topping."
