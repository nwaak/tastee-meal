ingredient_sections:
  - section_title: "Dough"
    items:
      - "1 tsp yeast"
      - "2 1/4 cups flour"
      - "1 1/2 tbsp powdered milk"
      - "1/2 tbsp sugar"
      - "1/2 tsp salt"
      - "1 tbsp vegetable oil"
      - "1 cup lukewarm water"

instructions:
  - title: "Mix Ingredients"
    text: "Pour ingredients into bread maker in order. Let the bread maker mix for the first cycle and dough to rise for at least 30 min."
  - title: "Preheat the Oven"
    text: "Preheat the oven to 425°F."
  - title: "Make Pizza"
    text: "Roll out dough and add favorite toppings."
  - title: "Bake"
    text: "Bake for 12 minutes."

notes: "If using a Ninja Combi, divide your mixed and raised pizza dough in half and freeze the rest for later. Roll out the dough and place it in the deep tray, then add your favorite sauce and toppings. Set the appliance to Pizza mode to preheat, insert the tray, and bake for 13 minutes until golden and bubbly."
