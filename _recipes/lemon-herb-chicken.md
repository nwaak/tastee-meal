prep_time: "15 mins"

total_time: "35 mins"
servings: "4 bowls"

ingredient_sections:
  - section_title: "Chicken & Marinade"
    items:
      - "1.5 lbs boneless skinless chicken breast"
      - "2 tbsp olive oil"
      - "1 lemon (zested and juiced)"
      - "3 cloves garlic, minced"
      - "1 tsp dried oregano"
  - section_title: "Bowl Base"
    items:
      - "1 cup dry quinoa, rinsed"
      - "2 cups chicken or vegetable broth"
      - "1 cup cherry tomatoes, halved"
      - "1/2 English cucumber, diced"

instructions:
  - title: "Marinate the Chicken"
    text: "Whisk together olive oil, lemon juice, lemon zest, minced garlic, oregano, salt, and black pepper in a bowl. Coat the chicken breasts evenly and let sit at room temperature for 15 minutes while starting the quinoa."
  - title: "Cook the Quinoa"
    text: "Combine rinsed quinoa and broth in a medium saucepan. Bring to a boil, reduce heat to low, cover, and simmer for 15 minutes until liquid is absorbed. Remove from heat and fluff with a fork."
  - title: "Sear the Chicken"
    text: "Heat a large skillet over medium-high heat. Sear chicken for 5–6 minutes per side until golden brown and internal temperature reaches 165°F (74°C). Transfer to a cutting board and rest for 5 minutes before slicing."
  - title: "Assemble Bowls"
    text: "Divide quinoa among four bowls. Top with sliced chicken, cherry tomatoes, and diced cucumber. Drizzle any remaining pan juices over top."

notes: "Keeps well in airtight glass containers for up to 4 days. Store the cucumbers and tomatoes separately if packing for lunch so they stay crisp when reheating the chicken and quinoa."
