title: "Cinnamon Rolls"
category: "breakfast"
author: "Laura"
description: "Small-batch homemade cinnamon rolls with a brown sugar and cinnamon filling, topped with cream cheese icing."
prep_time: ""
cook_time: "25-30 mins"
total_time: ""
servings: "6 rolls"
image: "https://images.unsplash.com/photo-1686207855146-c3ffe2166d40?auto=format&fit=crop&w=1200&q=80"

ingredient_sections:
  - section_title: "Dough"
    items:
      - "2 1/2 - 3 cups flour"
      - "2 tbsp + 2 tsp sugar"
      - "2 1/4 tsp yeast"
      - "1/2 tsp salt"
      - "3/4 cup water"
      - "3 tbsp butter"
      - "1/2 egg"
  - section_title: "Filling"
    items:
      - "2 tbsp butter"
      - "1/4 cup brown sugar"
      - "1/2 tbsp cinnamon"
  - section_title: "Icing"
    items:
      - "4 oz cream cheese"
      - "1 cup powdered sugar"
      - "2 tsp vanilla"

instructions:
  - title: "Mix Dry Ingredients"
    text: "Preheat the oven to 350°F. Mix 1 cup of the flour with the sugar, yeast, and salt."
  - title: "Heat Water & Butter"
    text: "Combine the water and butter and microwave for 45 seconds."
  - title: "Combine"
    text: "Pour the water and butter mixture into the flour mixture along with the egg."
  - title: "Add Flour"
    text: "Add 1 cup of flour and stir. Add 1/4 cup more flour and check whether the dough is still sticky."
  - title: "Rise & Roll Out"
    text: "Let the dough rise for 10 minutes, then roll it out."
  - title: "Add Filling"
    text: "Layer on the filling: 2 tbsp butter, 1/4 cup brown sugar, and 1/2 tbsp cinnamon."
  - title: "Bake"
    text: "Roll up and cut into 6 rolls. Bake at 350°F for 25-30 minutes."
  - title: "Ice"
    text: "Mix the cream cheese, powdered sugar, and vanilla together, and spread the icing over the rolls."
