title: "Chocolate Chip Muffins"
category: "breakfast"
author: "Laura"
description: "Simple homemade muffins with creamed butter and sugar, studded with chocolate chips."
prep_time: ""
cook_time: "20-25 mins"
total_time: ""
servings: "6 muffins"
image: "https://images.unsplash.com/photo-1472073818961-c5665e88ef83?auto=format&fit=crop&w=1200&q=80"

ingredient_sections:
  - section_title: "Batter"
    items:
      - "1/4 cup butter"
      - "1/2 cup sugar (white & brown)"
      - "1 egg"
      - "1 tsp baking powder"
      - "1/4 tsp salt"
      - "1/2 tsp vanilla"
      - "1 cup flour"
      - "1/4 cup milk"
  - section_title: "Mix-In"
    items:
      - "1/2 cup chocolate chips"

instructions:
  - title: "Preheat the Oven"
    text: "Preheat the oven to 425°F. After 7 minutes of baking, lower the temperature to 350°F."
  - title: "Cream Butter & Sugar"
    text: "Cream together the butter and sugar."
  - title: "Add Egg & Flavorings"
    text: "Beat in the egg, then add the baking powder, salt, and vanilla."
  - title: "Add Flour & Milk"
    text: "Add half of the flour and mix until combined. Add the milk to combine, then add the remaining flour."
  - title: "Fold in Chocolate Chips"
    text: "Fold in the chocolate chips."
  - title: "Bake"
    text: "Bake for 20-25 minutes total, lowering the oven from 425°F to 350°F after the first 7 minutes."
