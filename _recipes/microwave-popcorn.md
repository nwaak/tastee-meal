ingredient_sections:
  - section_title: "Popcorn & Seasoning"
    items:
      - "1/4 cup yellow popcorn"
      - "1 tablespoon butter"
      - "Salt, to taste"

instructions:
  - title: "Pop the Corn"
    text: "Place the yellow popcorn in your microwave-safe container. For a black silicone container, microwave for 1 minute and 45 seconds. For a glass container, microwave for 2 minutes and 15 seconds."
  - title: "Melt the Butter"
    text: "After the popcorn is popped, melt the butter in a small dish for 15 to 30 seconds."
  - title: "Combine"
    text: "Pour the melted butter over the warm popcorn, add salt to taste, and toss to coat."

notes: "Adjust microwave times depending on the exact wattage of your appliance and the type of container used."
