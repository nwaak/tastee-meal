description: "A classic homemade barbecue sauce combining ketchup, vinegar, brown sugar, and a blend of spices."
prep_time: "10 mins"
cook_time: "10 mins"
total_time: "20 mins"
servings: "4 cups"
image: "https://images.unsplash.com/photo-1767065703745-f3c33524605a?auto=format&fit=crop&w=1200&q=80"

ingredient_sections:
  - section_title: "Sauce Ingredients"
    items:
      - "2 cups ketchup"
      - "1 1/2 cups water"
      - "4 Tbs vinegar"
      - "6 Tbs brown sugar"
      - "2 tsp salt"
      - "2 tsp paprika"
      - "1 tsp celery salt"
      - "1 1/2 tsp coarse black pepper"
      - "1/2 tsp MSG"
      - "2 Tbs flour"
      - "1/2 tsp onion salt"
      - "1/4 tsp garlic salt"

instructions:
  - title: "Combine Dry Ingredients"
    text: "Combine all dry ingredients and mix well, then set aside."
  - title: "Combine Wet Ingredients"
    text: "In a large saucepan, combine ketchup, water, and vinegar."
  - title: "Heat and Thicken"
    text: "Bring to a near boil, then remove from heat. Add the dry ingredients and mix well."

notes: "Let cool before storing in an airtight container in the refrigerator."
