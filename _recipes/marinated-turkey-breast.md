ingredient_sections:
  - section_title: "Turkey & Rub"
    items:
      - "3 lb turkey breast"
      - "2 cloves garlic, minced"
      - "1 tbsp chopped fresh basil"
      - "1/2 tsp black pepper"
  - section_title: "Marinade"
    items:
      - "1/4 cup vegetable oil"
      - "1/4 cup soy sauce"
      - "2 tbsp lemon juice"
      - "1 tbsp brown sugar"

instructions:
  - title: "Rub the Turkey"
    text: "Rub the garlic, basil, and pepper onto the turkey."
  - title: "Make the Marinade"
    text: "Whisk together the vegetable oil, soy sauce, lemon juice, and brown sugar, then pour it over the turkey."
  - title: "Marinate"
    text: "Marinate for at least 4 hours."
  - title: "Grill"
    text: "Grill for 15 minutes on each side."
